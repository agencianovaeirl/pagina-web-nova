-- CREAR TABLA galeria_carrusles EN SUPABASE
CREATE TABLE IF NOT EXISTS public.galeria_carrusles (
    id TEXT PRIMARY KEY,
    seccion TEXT DEFAULT 'maluma',
    categoria TEXT,
    url TEXT,
    titulo TEXT,
    orden INTEGER DEFAULT 0,
    data JSONB,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- HABILITAR SEGURIDAD RLS Y PERMISOS PÚBLICOS
ALTER TABLE public.galeria_carrusles ENABLE ROW LEVEL SECURITY;
DROP POLICY IF EXISTS "Lectura publica galeria_carrusles" ON public.galeria_carrusles;
CREATE POLICY "Lectura publica galeria_carrusles" ON public.galeria_carrusles FOR SELECT USING (true);
DROP POLICY IF EXISTS "Escritura publica galeria_carrusles" ON public.galeria_carrusles;
CREATE POLICY "Escritura publica galeria_carrusles" ON public.galeria_carrusles FOR ALL USING (true);
