Código SQL para crear la tabla nova_propiedades en Supabase: 
CREATE TABLE IF NOT EXISTS nova_propiedades (
    id TEXT PRIMARY KEY,
    title TEXT NOT NULL,
    location TEXT NOT NULL,
    location_key TEXT DEFAULT 'cusco',
    type TEXT DEFAULT 'terreno',
    price TEXT NOT NULL,
    price_num NUMERIC DEFAULT 0,
    area TEXT NOT NULL,
    image TEXT NOT NULL,
    image2 TEXT,
    image3 TEXT,
    feat1_icon TEXT DEFAULT 'verified',
    feat1_text TEXT DEFAULT 'Título SUNARP',
    feat2_icon TEXT DEFAULT 'bolt',
    feat2_text TEXT DEFAULT 'Servicios Listos',
    detailed_desc TEXT,
    coords TEXT DEFAULT '-13.3039,-72.1158',
    map_zoom INTEGER DEFAULT 13,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- Políticas RLS para lectura pública y guardado:
ALTER TABLE nova_propiedades ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Lectura pública de propiedades" ON nova_propiedades FOR SELECT USING (true);
CREATE POLICY "Escritura autorizada de propiedades" ON nova_propiedades FOR ALL USING (true);
Los cambios se guardan automáticamente y se actualizan en el catálogo y mapa interactivo.
